cmake_minimum_required(VERSION 3.14)
project(TrabalhoGrauA LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Estas dependencias pertencem somente ao jogo, inclusive no build pela raiz.
include(FetchContent)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(glfw
    GIT_REPOSITORY https://github.com/glfw/glfw.git
    GIT_TAG 3.4
    GIT_SHALLOW TRUE
)
FetchContent_Declare(glm
    GIT_REPOSITORY https://github.com/g-truc/glm.git
    GIT_TAG 1.0.1
    GIT_SHALLOW TRUE
)
FetchContent_MakeAvailable(glfw glm)
find_package(OpenGL REQUIRED)

add_library(grau_a_glad STATIC third_party/glad/src/glad.c)
target_include_directories(grau_a_glad PUBLIC third_party/glad/include)

option(GRAU_A_AUDIO "Habilitar audio do jogo com miniaudio" ON)
add_library(grau_a_audio STATIC Audio.cpp)
target_include_directories(grau_a_audio PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(GRAU_A_AUDIO)
    find_package(Threads REQUIRED)
    add_library(grau_a_miniaudio STATIC third_party/miniaudio/implementation.c)
    target_include_directories(grau_a_miniaudio SYSTEM PUBLIC third_party/miniaudio)
    target_link_libraries(grau_a_miniaudio PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
    if(UNIX)
        target_link_libraries(grau_a_miniaudio PUBLIC m)
    endif()
    target_link_libraries(grau_a_audio PRIVATE grau_a_miniaudio)
    target_compile_definitions(grau_a_audio PRIVATE GRAU_A_AUDIO)
endif()

add_executable(TrabalhoGrauA main.cpp Game.cpp Config.cpp Hud.cpp Player.cpp World.cpp Encounters.cpp Combat.cpp Renderer2D.cpp Texture.cpp RenderCheck.cpp)
target_include_directories(TrabalhoGrauA SYSTEM PRIVATE third_party/stb)
target_link_libraries(TrabalhoGrauA PRIVATE grau_a_glad glfw glm::glm OpenGL::GL grau_a_audio)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(TrabalhoGrauA PRIVATE -Wall -Wextra -Wpedantic)
endif()

get_filename_component(default_assets
    "${CMAKE_CURRENT_SOURCE_DIR}/../../../assets/TrabalhoGrauA" ABSOLUTE)
set(TRABALHO_GRAU_A_ASSET_DIR "${default_assets}" CACHE PATH "Assets finais do jogo")
if(NOT IS_DIRECTORY "${TRABALHO_GRAU_A_ASSET_DIR}")
    message(FATAL_ERROR "Assets nao encontrados: ${TRABALHO_GRAU_A_ASSET_DIR}")
endif()
file(TO_CMAKE_PATH "${TRABALHO_GRAU_A_ASSET_DIR}" game_assets)
target_compile_definitions(TrabalhoGrauA PRIVATE
    TRABALHO_GRAU_A_ASSET_DIR="${game_assets}")

include(CTest)
if(BUILD_TESTING)
    add_executable(TrabalhoGrauA_ConfigTests tests/ConfigTests.cpp Config.cpp Player.cpp Encounters.cpp)
    target_include_directories(TrabalhoGrauA_ConfigTests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    add_test(NAME Configuration COMMAND TrabalhoGrauA_ConfigTests)
    if(GRAU_A_AUDIO)
        add_executable(TrabalhoGrauA_AudioTests tests/AudioTests.cpp)
        target_link_libraries(TrabalhoGrauA_AudioTests PRIVATE grau_a_audio)
        add_test(NAME Audio COMMAND TrabalhoGrauA_AudioTests "${game_assets}")
    endif()
    add_executable(TrabalhoGrauA_PlayerTests tests/PlayerTests.cpp Player.cpp)
    target_include_directories(TrabalhoGrauA_PlayerTests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(TrabalhoGrauA_PlayerTests PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    add_test(NAME PlayerLogic COMMAND TrabalhoGrauA_PlayerTests)
    add_executable(TrabalhoGrauA_WorldTests tests/WorldTests.cpp)
    target_include_directories(TrabalhoGrauA_WorldTests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(TrabalhoGrauA_WorldTests PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    add_test(NAME WorldScrolling COMMAND TrabalhoGrauA_WorldTests)
    add_executable(TrabalhoGrauA_EncounterTests tests/EncounterTests.cpp Encounters.cpp)
    target_include_directories(TrabalhoGrauA_EncounterTests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(TrabalhoGrauA_EncounterTests PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    add_test(NAME Encounters COMMAND TrabalhoGrauA_EncounterTests)
    add_executable(TrabalhoGrauA_CombatTests tests/CombatTests.cpp Combat.cpp Player.cpp Encounters.cpp)
    target_include_directories(TrabalhoGrauA_CombatTests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(TrabalhoGrauA_CombatTests PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    add_test(NAME Combat COMMAND TrabalhoGrauA_CombatTests)
    add_executable(TrabalhoGrauA_ProgressionTests tests/ProgressionTests.cpp Combat.cpp Player.cpp Encounters.cpp)
    target_include_directories(TrabalhoGrauA_ProgressionTests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(TrabalhoGrauA_ProgressionTests PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    add_test(NAME Progression COMMAND TrabalhoGrauA_ProgressionTests)
endif()
